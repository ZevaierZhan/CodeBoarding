import hashlib
import json
import subprocess
from pathlib import Path
from string import Template

ROOT = Path.cwd()
OUT = ROOT / "examples/architecture_model/codeboarding"
PREFIX = ["uv", "run", "python", "-m", "architecture_model", "model", "--file", "examples/architecture_model/codeboarding/model.json"]
records = []


def command(arguments, artifact):
    argv = PREFIX + arguments
    result = subprocess.run(argv, cwd=ROOT, capture_output=True, text=True, encoding="utf-8")
    records.append({"argv": argv, "cwd": str(ROOT), "exit_code": result.returncode, "stdout": result.stdout, "stderr": result.stderr})
    (OUT / "03-commands.json").write_text(json.dumps(records, indent=2), encoding="utf-8")
    (OUT / artifact).write_text(result.stdout, encoding="utf-8")
    parsed = json.loads(result.stdout)
    if result.returncode or not parsed.get("ok"):
        raise RuntimeError(f"Command failed; inspect {artifact}")
    return parsed["result"]


before = command(["read"], "03-before-read.json")["model"]
scope = json.loads((OUT / "scope.json").read_text(encoding="utf-8"))
hashes = {path: hashlib.sha256((ROOT / path).read_bytes()).hexdigest() for path in scope["included"]}
assert hashes == scope["included"] == before["facts"][0]["source"]["files"]
(OUT / "03-source-check.json").write_text(json.dumps({"all_manifest_bytes_match": True, "files": hashes, "snapshot_id": before["facts"][0]["id"], "source_version": before["facts"][0]["source"]["digest"]}, indent=2), encoding="utf-8")
assert before["revision"] == 2
command(["maintenance"], "03-maintenance-before.json")


def claim(text, bindings, inference=False):
    return {"kind": "inference" if inference else "code_fact", "explanation": text, "binding_ids": bindings, "assumptions": ["An external Agent chooses whether to apply after reviewing preview output."] if inference else [], "issues": []}


def step(identity, components, bindings, text, inference=False):
    return {"id": identity, "component_ids": components, "binding_ids": bindings, "claim": claim(text, bindings, inference)}


steps = [
    step("patch-submit", ["cli", "store"], ["e-cli-run", "e-store-json"], "run accepts --patch for validate/diff/apply; load_json rejects malformed JSON and duplicate keys before dispatch into preview/apply. apply also requires --expected-revision."),
    step("patch-preview", ["store", "validation"], ["e-store-preview", "e-store-candidate", "e-store-patched", "e-store-revision", "e-validate"], "validate --patch and diff --patch both call preview: read and validate current model, build a candidate, calculate impact and append a candidate revision in memory. The response reports committed:false; no save or lock is performed."),
    step("patch-agent-review", ["cli", "store"], ["e-cli-run", "e-store-preview", "e-store-apply"], "The external Agent inspects affected objects/scenarios and may submit apply in a separate invocation. Preview does not reserve its revision, and review/retry policy belongs to the caller.", True),
    step("patch-lock-read", ["store", "validation"], ["e-store-apply", "e-store-commit", "e-store-read"], "apply delegates to commit with a candidate-builder callback. Under the file lock (30-second timeout), commit rereads and validates the saved model before comparing expected_revision."),
    step("patch-conflict", ["store", "cli"], ["e-store-commit", "e-cli-main"], "A revision mismatch raises REVISION_CONFLICT before invoking the builder or writing. main returns ok:false and exit 1; the caller must reread and reconcile before resubmission."),
    step("patch-candidate", ["store", "validation"], ["e-store-candidate", "e-store-patched", "e-store-revision", "e-store-diff", "e-validate"], "The builder parses Patch, copies current model data, derives maintenance observations and processes operations sequentially. Facts are append-only, IDs cannot change, create cannot overwrite reserved IDs, and update/delete require existing objects. Update fields replace whole values. Complete-model validation checks references, hierarchy and evidence. revision_candidate preserves binding history, derives impact, increments revision, appends author/summary/timestamp/affected IDs and validates again."),
    step("patch-atomic-save", ["store"], ["e-store-commit", "e-store-atomic"], "Still holding the lock, commit prepares the committed:true result and atomic_write serializes the candidate into a temporary file beside the model, flushes and fsyncs, then os.replace publishes it. os.replace is the commit point; finally removes a remaining temporary file."),
    step("patch-rejected", ["cli", "store", "validation"], ["e-cli-main", "e-store-commit", "e-store-atomic", "e-validate"], "ModelError or caught IO/lock errors become ok:false JSON and exit 1 if output delivery succeeds. LOCK_TIMEOUT and pre-replace validation/save failures do not commit this patch. Errors after os.replace or output delivery require reading the model to establish outcome."),
    step("patch-response", ["cli"], ["e-cli-main", "e-store-commit"], "After commit returns, main writes and flushes ok:true JSON with committed:true, revision and diff, then exits 0 if delivery succeeds."),
    step("patch-output-uncertain", ["cli"], ["e-cli-main", "e-store-atomic"], "If JSON stdout write/flush raises OSError or UnicodeError, main attempts an stderr diagnostic and returns 2. This does not roll back os.replace; an observed delivery failure alone does not establish whether apply committed."),
    step("patch-read-recovery", ["cli", "store", "validation"], ["e-cli-run", "e-store-read", "e-store-scope"], "A separate read invocation returns the complete validated saved model when no scope is supplied. Inspect revision and explanation before retrying after conflict or uncertain delivery; read itself does not retry apply."),
